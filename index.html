GuardDuty Alert Triage — Agentic DAG
Multi-source correlation across CloudTrail, Okta, CrowdStrike EDR, and GitHub Enterprise. Mean time to triage reduced from 30 min manual to <3 min agent-assisted.
# GuardDuty Triage — Agentic Workflow (pseudocode) async def triage_guardduty_alert(finding_id: str) -> TriageResult: # Phase 0-1: Intake + NGSIEM lookup finding = await ngsiem_lookup(finding_id) principal, alert_ip, timestamp = extract_identity(finding) # Branch: Kubernetes vs Identity Center vs Standard if "Kubernetes" in finding.type: return await kubernetes_triage_path(finding) elif principal == "HIDDEN_DUE_TO_SECURITY_REASONS": return await identity_center_path(finding) # Phases 2-8: Parallel fan-out across all data sources cloudtrail, okta, crowdstrike, baseline, github, ssm = await asyncio.gather( cloudtrail_timeline(principal, timestamp), okta_sso_timeline(principal, timestamp), crowdstrike_device_history(principal, alert_ip), consolein_baseline(principal, exclude_day=timestamp), github_activity(principal, timestamp), ssm_session_activity(principal, timestamp), ) # Phase 10: Score and verdict score = risk_score(cloudtrail, okta, crowdstrike, baseline, github, ssm) verdict = "HIGH" if score >= 5 else "MEDIUM" if score >= 2 else "LOW" # Phases 11-12: Writeup + Jira + Slack report = write_investigation_file(finding, verdict, evidence) await asyncio.gather( jira_create_ticket(verdict, report), slack_post("#lt-brainstorm", verdict, report), ) return TriageResult(verdict=verdict, score=score, report=report)

# Supply Chain Triage — Agentic Workflow (pseudocode) async def triage_package_alert(alert: PackageAlert) -> Verdict: # Skip if already investigated if has_existing_verdict(alert.package, alert.version): return get_cached_verdict(alert) # Clone and scan in parallel across all engines osv, semgrep, yara, gitleaks, nova = await asyncio.gather( osv_scan(alert.package), # known CVEs semgrep_scan(repo_path, rules=85), # malicious patterns yara_scan(repo_path, rules=27), # supply chain signatures gitleaks_scan(repo_path), # hardcoded secrets nova_scan(repo_path), # AI/prompt injection ) verdict = aggregate_verdict(osv, semgrep, yara, gitleaks, nova) # Output: Slack + GitHub Discussion + bot-team-kb commit await asyncio.gather( post_slack("#lt-brainstorm", verdict), post_github_discussion(verdict), commit_investigation_file("bot-team-kb/security-investigations/", verdict), ) return verdict
